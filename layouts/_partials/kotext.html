{{- /* Fixes line breaks in Korean text. On English pages it returns the text unchanged. Chromium browsers break Korean lines at these points, even with word-break: keep-all. This partial adds a word joiner (U+2060, invisible) at each point to stop the break: - before and after a middle dot, so that '·' never starts a line ('타이머·PWM') - between Hangul and an opening parenthesis ('제어기(MFC)') - between a closing parenthesis and Hangul, so that a particle never starts a line ('주기)을') Source files need only normal characters. Takes a string and returns a string, or takes HTML (.Content, markdownify output) and returns HTML. Example: {{ partial "kotext.html" .Title }} */ -}} {{- $out := . -}} {{- if eq site.Language.Lang "ko" }} {{- $s := replace . "·" "⁠·⁠" }} {{- $s = replaceRE `([가-힣])\(` "${1}⁠(" $s }} {{- $s = replaceRE `\)([가-힣])` ")⁠${1}" $s }} {{- if eq (printf "%T" .) "template.HTML" }}{{ $out = $s | safeHTML }}{{ else }}{{ $out = $s }}{{ end }} {{- end -}} {{- return $out -}}
